package com.cortex.instead;

import android.annotation.SuppressLint;
import android.app.Activity;
import android.content.Intent;
import android.database.Cursor;
import android.net.Uri;
import android.os.Bundle;
import android.os.Environment;
import android.provider.OpenableColumns;
import android.view.KeyEvent;
import android.webkit.JavascriptInterface;
import android.webkit.ValueCallback;
import android.webkit.WebChromeClient;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.widget.Toast;
import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.OutputStream;

public class MainActivity extends Activity {

    public static MainActivity instance;
    private WebView web;
    private ValueCallback<Uri[]> fileCallback;
    private static final int PICK_FILE = 42;

    @SuppressLint("SetJavaScriptEnabled")
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        instance = this;
        web = new WebView(this);
        setContentView(web);
        web.getSettings().setJavaScriptEnabled(true);
        web.getSettings().setDomStorageEnabled(true);
        web.getSettings().setMediaPlaybackRequiresUserGesture(false);
        web.getSettings().setAllowFileAccess(true);
        web.addJavascriptInterface(new Bridge(), "Android");
        web.setWebViewClient(new WebViewClient());
        web.setWebChromeClient(new WebChromeClient() {
            @Override
            public boolean onShowFileChooser(WebView v, ValueCallback<Uri[]> cb, FileChooserParams params) {
                fileCallback = cb;
                try {
                    Intent i = new Intent(Intent.ACTION_GET_CONTENT);
                    i.addCategory(Intent.CATEGORY_OPENABLE);
                    i.setType("audio/*");
                    startActivityForResult(Intent.createChooser(i, "Pick a track"), PICK_FILE);
                } catch (Exception e) {
                    fileCallback = null;
                    return false;
                }
                return true;
            }
        });
        web.loadUrl("file:///android_asset/app/index.html");
    }

    /** Run JS on the UI thread (called from the playback service). */
    public void pushJs(final String js) {
        runOnUiThread(new Runnable() {
            public void run() {
                if (web != null) web.evaluateJavascript(js, null);
            }
        });
    }

    static String json(String s) {
        return "\"" + s.replace("\\", "\\\\").replace("\"", "\\\"")
                .replace("\n", "\\n").replace("\r", "").replace("'", "\\'") + "\"";
    }

    class Bridge {

        @JavascriptInterface
        public void play(String file, String title) {
            PlayerService.start(MainActivity.this, "asset:" + file, title);
        }

        @JavascriptInterface
        public void playPath(String uri, String title) {
            PlayerService.start(MainActivity.this, uri, title);
        }

        @JavascriptInterface
        public void pause() { PlayerService.pausePlay(); }

        @JavascriptInterface
        public void resume() { PlayerService.resumePlay(); }

        @JavascriptInterface
        public void stop() { PlayerService.stopPlay(); }

        @JavascriptInterface
        public void share(String text) {
            try {
                Intent i = new Intent(Intent.ACTION_SEND);
                i.setType("text/plain");
                i.putExtra(Intent.EXTRA_TEXT, text);
                startActivity(Intent.createChooser(i, "Share via"));
            } catch (Exception e) {
                toast("No app to share with");
            }
        }

        @JavascriptInterface
        public void download(String file) {
            try {
                File dir = getExternalFilesDir(Environment.DIRECTORY_MUSIC);
                if (dir == null) dir = getFilesDir();
                if (!dir.exists()) dir.mkdirs();
                File out = new File(dir, file);
                InputStream in = getAssets().open("app/audio/" + file);
                OutputStream os = new FileOutputStream(out);
                byte[] buf = new byte[8192];
                int n;
                while ((n = in.read(buf)) > 0) os.write(buf, 0, n);
                os.close();
                in.close();
                toast("Saved offline: " + out.getAbsolutePath());
            } catch (Exception e) {
                toast("Download failed: " + e.getMessage());
            }
        }

        @JavascriptInterface
        public void toast(final String msg) {
            runOnUiThread(new Runnable() {
                public void run() {
                    Toast.makeText(MainActivity.this, msg, Toast.LENGTH_SHORT).show();
                }
            });
        }

        @JavascriptInterface
        public void exitApp() {
            runOnUiThread(new Runnable() {
                public void run() { finish(); }
            });
        }
    }

    @Override
    protected void onActivityResult(int requestCode, int resultCode, Intent data) {
        if (requestCode == PICK_FILE && fileCallback != null) {
            Uri[] uris = null;
            if (resultCode == RESULT_OK && data != null && data.getData() != null) {
                uris = new Uri[]{ data.getData() };
            }
            fileCallback.onReceiveValue(uris);
            fileCallback = null;
            if (uris != null) {
                String name = "My Track";
                try {
                    Cursor c = getContentResolver().query(uris[0], null, null, null, null);
                    if (c != null && c.moveToFirst()) {
                        int i = c.getColumnIndex(OpenableColumns.DISPLAY_NAME);
                        if (i >= 0 && c.getString(i) != null) name = c.getString(i);
                    }
                    if (c != null) c.close();
                } catch (Exception ignored) { }
                final String uri = uris[0].toString();
                final String nm = name;
                pushJs("window.app.onFilePicked(" + json(uri) + "," + json(nm) + ")");
            }
        } else {
            super.onActivityResult(requestCode, resultCode, data);
        }
    }

    @Override
    public boolean onKeyDown(int keyCode, KeyEvent event) {
        if (keyCode == KeyEvent.KEYCODE_BACK) {
            pushJs("window.app.back()");
            return true;
        }
        return super.onKeyDown(keyCode, event);
    }

    @Override
    protected void onDestroy() {
        instance = null;
        super.onDestroy();
    }
}

package com.cortex.instead;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.content.res.AssetFileDescriptor;
import android.media.MediaPlayer;
import android.net.Uri;
import android.os.Build;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;

/**
 * Foreground playback service — keeps music running when the app is minimized,
 * shows a notification with play/pause, and pushes progress to the WebView UI.
 */
public class PlayerService extends Service {

    public static PlayerService instance;
    private MediaPlayer mp;
    private String source = "", title = "";
    private final Handler h = new Handler(Looper.getMainLooper());

    private final Runnable tick = new Runnable() {
        public void run() {
            pushProgress();
            h.postDelayed(this, 500);
        }
    };

    @Override
    public void onCreate() {
        super.onCreate();
        instance = this;
        if (Build.VERSION.SDK_INT >= 26) {
            NotificationManager nm = (NotificationManager) getSystemService(NOTIFICATION_SERVICE);
            nm.createNotificationChannel(new NotificationChannel(
                    "instead", "Playback", NotificationManager.IMPORTANCE_LOW));
        }
    }

    public static void start(Context c, String source, String title) {
        Intent i = new Intent(c, PlayerService.class);
        i.putExtra("source", source);
        i.putExtra("title", title);
        try {
            if (Build.VERSION.SDK_INT >= 26) c.startForegroundService(i);
            else c.startService(i);
        } catch (Exception e) {
            c.startService(i);
        }
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        if (intent != null && "toggle".equals(intent.getAction())) {
            togglePlayback();
            return START_NOT_STICKY;
        }
        if (intent != null && intent.getStringExtra("source") != null) {
            play(intent.getStringExtra("source"), intent.getStringExtra("title"));
        }
        startForeground(1, notif());
        h.removeCallbacks(tick);
        h.post(tick);
        return START_NOT_STICKY;
    }

    private void togglePlayback() {
        if (mp == null) return;
        try {
            if (mp.isPlaying()) mp.pause();
            else mp.start();
        } catch (Exception ignored) { }
        updateNotification();
        if (MainActivity.instance != null) {
            MainActivity.instance.pushJs("window.app.onProgress(" + safePos() + "," + safeDur() + ")");
        }
    }

    private void play(String src, String ttl) {
        this.source = src == null ? "" : src;
        this.title = ttl == null ? "" : ttl;
        try {
            if (mp == null) {
                mp = new MediaPlayer();
                mp.setOnCompletionListener(new MediaPlayer.OnCompletionListener() {
                    public void onCompletion(MediaPlayer m) {
                        if (MainActivity.instance != null)
                            MainActivity.instance.pushJs("window.app.onEnded()");
                    }
                });
            } else {
                mp.reset();
            }
            if (source.startsWith("asset:")) {
                AssetFileDescriptor a = getAssets().openFd("app/audio/" + source.substring(6));
                mp.setDataSource(a.getFileDescriptor(), a.getStartOffset(), a.getLength());
                a.close();
            } else {
                mp.setDataSource(this, Uri.parse(source));
            }
            mp.prepare();
            mp.start();
        } catch (Exception e) {
            if (MainActivity.instance != null)
                MainActivity.instance.pushJs("window.app.toastMsg && window.app.toastMsg(" +
                        MainActivity.json("Could not play this file: " + e.getMessage()) + ")");
        }
        updateNotification();
    }

    public static void pausePlay() {
        if (instance != null && instance.mp != null) {
            try { instance.mp.pause(); instance.updateNotification(); } catch (Exception ignored) { }
        }
    }

    public static void resumePlay() {
        if (instance != null && instance.mp != null) {
            try { instance.mp.start(); instance.updateNotification(); } catch (Exception ignored) { }
        }
    }

    public static void stopPlay() {
        if (instance != null) instance.stopSelf();
    }

    private int safePos() { try { return mp.getCurrentPosition(); } catch (Exception e) { return 0; } }
    private int safeDur() { try { return mp.getDuration(); } catch (Exception e) { return 0; } }

    private void pushProgress() {
        if (mp == null || MainActivity.instance == null) return;
        try {
            MainActivity.instance.pushJs("window.app.onProgress(" + safePos() + "," + safeDur() + ")");
        } catch (Exception ignored) { }
    }

    private Notification notif() {
        Intent open = new Intent(this, MainActivity.class);
        open.setFlags(Intent.FLAG_ACTIVITY_SINGLE_TOP);
        PendingIntent pOpen = PendingIntent.getActivity(this, 1, open,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);

        Intent t = new Intent(this, PlayerService.class);
        t.setAction("toggle");
        PendingIntent pToggle = PendingIntent.getService(this, 2, t,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);

        Notification.Builder b = Build.VERSION.SDK_INT >= 26
                ? new Notification.Builder(this, "instead")
                : new Notification.Builder(this);
        b.setContentTitle("INSTEAD")
         .setContentText("♪ " + title + " — playing in background")
         .setSmallIcon(android.R.drawable.ic_media_play)
         .setOngoing(true)
         .setContentIntent(pOpen)
         .addAction(new Notification.Action.Builder(
                 android.R.drawable.ic_media_pause, "Pause / Play", pToggle).build());
        return b.build();
    }

    private void updateNotification() {
        try {
            NotificationManager nm = (NotificationManager) getSystemService(NOTIFICATION_SERVICE);
            nm.notify(1, notif());
        } catch (Exception ignored) { }
    }

    @Override
    public void onDestroy() {
        h.removeCallbacks(tick);
        if (mp != null) { try { mp.release(); } catch (Exception ignored) { } mp = null; }
        instance = null;
        super.onDestroy();
    }

    @Override
    public IBinder onBind(Intent intent) { return null; }
}
