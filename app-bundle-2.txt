function songRow(song, opts = {}) {
  const isCurrent = state.current === song.id;
  const saved = state.downloaded.includes(song.id);
  const row = document.createElement('div');
  row.className = 'song-row';
  row.innerHTML = `

    <div class="meta">

      <div class="artist">${song.artist} · ${song.genre}${saved ? ' · ✓ saved' : ''}</div>

function renderSearch() {
  const q = ($('#search-input').value || '').toLowerCase().trim();
  const chips = ['All', ...new Set(allSongs().map(s => s.genre))];
  const chipEl = $('#genre-chips');
  chipEl.innerHTML = '';
  chips.forEach(c => {
    const d = document.createElement('div');
    d.className = 'chip' + ((state.chip || 'All') === c ? ' active' : '');
    d.textContent = c;
    d.onclick = () => { state.chip = c; renderSearch(); };
    chipEl.appendChild(d);
  });
  let res = allSongs();
  if (state.chip && state.chip !== 'All') res = res.filter(s => s.genre === state.chip);
  if (q) res = res.filter(s => (s.title + ' ' + s.artist + ' ' + s.genre).toLowerCase().includes(q));
  $('#search-count').textContent = q ? `${res.length} result${res.length === 1 ? '' : 's'} for “${q}”` : 'All songs';
  const list = $('#search-list');
  list.innerHTML = '';
  if (!res.length) { list.innerHTML = '<div class="empty">No songs found — try “monsoon”, “lo-fi”, or an artist name.</div>'; return; }
  res.forEach(s => list.appendChild(songRow(s)));
}

/* ---------------- playback ---------------- */
function playSong(id) {
  const s = songById(id);
  if (!s) return;
  state.current = id;
  state.playing = true;
  clearInterval(state.simTimer);
  state.posMs = 0; state.durMs = 0;
  if (s.path && AND) AND.playPath(s.path, s.title);
  else if (AND) AND.play(s.file, s.title);
  else { state.durMs = 21000; simulate(); }         // browser demo mode
  show('player');
  renderPlayer();
}

function simulate() {
  clearInterval(state.simTimer);
  state.simTimer = setInterval(() => {
    if (!state.playing) return;
    state.posMs += 500;
    if (state.posMs >= state.durMs) onEnded();
    else onProgress(state.posMs, state.durMs);
  }, 500);
}

function togglePlay() {
  const s = songById(state.current);
  if (!s) { playSong(SONGS[0].id); return; }
  state.playing = !state.playing;
  if (state.playing) { if (AND) AND.resume(); else if (!state.durMs) { state.durMs = 21000; } simulate(); }
  else { if (AND) AND.pause(); clearInterval(state.simTimer); }
  updatePlayIcon();
  renderHome();
}

function nextSong(dir = 1) {
  const list = allSongs();
  const i = list.findIndex(s => s.id === state.current);
  const n = list[(i + dir + list.length) % list.length];
  playSong(n.id);
}

function onEnded() {
  const s = songById(state.current);
  if (s) s.plays = 'just played';
  nextSong(1);
}
window.app.onEnded = onEnded;

function fmt(ms) {
  const t = Math.floor(ms / 1000);
  return Math.floor(t / 60) + ':' + String(t % 60).padStart(2, '0');
}
function onProgress(pos, dur) {
  state.posMs = pos; state.durMs = dur;
  const pct = dur ? Math.min(100, (pos / dur) * 100) : 0;
  $('#seek-fill').style.width = pct + '%';
  $('#t-now').textContent = fmt(pos);
  $('#t-dur').textContent = '-' + fmt(Math.max(0, dur - pos));
}
window.app.onProgress = onProgress;

function runUpProgress(ms) {
  const t0 = performance.now();
  const tick = () => {
    const p = Math.min(100, Math.round(((performance.now() - t0) / ms) * 100));
    $('#up-fill').style.width = p + '%';
    $('#up-pct').textContent = p + '%';
    if (p < 100) requestAnimationFrame(tick);
    else { $('#up-status').textContent = 'Ready to publish'; $('#up-pct').textContent = '✓'; }
  };
  requestAnimationFrame(tick);
}

function finishUploadDemo(name) {

#!/usr/bin/env python3
"""Generates the launcher icon (PNG) and sample audio (WAV) for the INSTEAD app.
Run before building. Pure stdlib for audio; Pillow for the icon."""
import math
import os
import random
import struct
import wave

ROOT = os.path.dirname(os.path.abspath(__file__))
AUDIO_DIR = os.path.join(ROOT, 'app', 'src', 'main', 'assets', 'app', 'audio')
ICON_PATH = os.path.join(ROOT, 'app', 'src', 'main', 'res', 'mipmap-xxxhdpi', 'ic_launcher.png')
SR = 22050

# ---------------- audio ----------------
NOTES = {
    'C3': 130.81, 'D3': 146.83, 'E3': 164.81, 'F3': 174.61, 'G3': 196.00, 'A3': 220.00, 'B3': 246.94,
    'C4': 261.63, 'D4': 293.66, 'E4': 329.63, 'F4': 349.23, 'G4': 392.00, 'A4': 440.00, 'B4': 493.88,
    'C5': 523.25, 'D5': 587.33, 'E5': 659.25, 'F5': 698.46, 'G5': 783.99, 'A5': 880.00, 'B5': 987.77,
}

def render_track(chords, melody, bpm, bars, seed, wave_shape='sine', swing=0.0):
    rng = random.Random(seed)
    beat = 60.0 / bpm
    total = int(SR * beat * bars * 4)
    buf = [0.0] * total

    def add(start, freq, dur, vol, shape):
        s = int(start * SR)
        n = int(dur * SR)
        for i in range(n):
            if s + i >= total: break
            t = i / SR
            ph = t * freq
            if shape == 'square':
                v = 1.0 if math.sin(2 * math.pi * ph) >= 0 else -1.0
            elif shape == 'tri':
                v = 2.0 * (ph % 1.0) - 1.0
            else:
                v = math.sin(2 * math.pi * ph)
            env = min(1.0, t * 20, (dur - t) * 12)
            buf[s + i] += v * vol * env

    for bar in range(bars):
        chord = chords[bar % len(chords)]
        t0 = bar * beat * 4
        for f in chord:
            add(t0, NOTES[f], beat * 3.9, 0.16, 'tri')

    for bar in range(bars):
        t0 = bar * beat * 4
        for step in range(8):
            if rng.random() < 0.28: continue
            f = melody[(bar * 8 + step) % len(melody)]
            swing_off = (beat * 0.5 * swing) if step % 2 == 1 else 0.0
            add(t0 + step * beat * 0.5 + swing_off, NOTES[f], beat * 0.46, 0.34, wave_shape)

    for bar in range(bars):
        chord = chords[bar % len(chords)]
        t0 = bar * beat * 4
        for beat_i in range(4):
            add(t0 + beat_i * beat, NOTES[chord[0]] / 2, beat * 0.9, 0.22, 'sine')

    for bar in range(bars):
        for step in range(8):
            t = int((bar * beat * 4 + step * beat * 0.5) * SR)
            for k in range(60):
                if t + k < total:
                    buf[t + k] += (rng.random() * 2 - 1) * 0.05 * (1 - k / 60)

    fade = int(SR * 1.2)
    for i in range(fade):
        j = total - 1 - i
        buf[j] *= i / fade
    peak = max(1e-6, max(abs(x) for x in buf))
    scale = 0.82 / peak
    return [x * scale for x in buf]

def write_wav(path, samples):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with wave.open(path, 'w') as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(SR)
        w.writeframes(b''.join(struct.pack('<h', max(-32767, min(32767, int(x * 32767)))) for x in samples))
    print('wrote', path, os.path.getsize(path) // 1024, 'KB')
